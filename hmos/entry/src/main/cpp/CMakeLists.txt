cmake_minimum_required(VERSION 3.16)
project(morrow_hmos)
set(CMAKE_CXX_STANDARD 17)
set(RUST_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/rust/${OHOS_ARCH}/libmorrow_hmos.a")
if(NOT EXISTS "${RUST_ARCHIVE}")
  message(FATAL_ERROR "Run scripts/build-rust.ps1 -Abi ${OHOS_ARCH} before building the HAP")
endif()
add_library(morrow_rust STATIC IMPORTED)
set_target_properties(morrow_rust PROPERTIES IMPORTED_LOCATION "${RUST_ARCHIVE}")
add_library(morrow SHARED bridge.cpp)
target_link_libraries(morrow PRIVATE morrow_rust libace_napi.z.so libace_ndk.z.so libhilog_ndk.z.so uv dl m)
target_link_options(morrow PRIVATE "-Wl,--exclude-libs,ALL")
